```php
<?php

namespace App\Models\Marketplace;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Str;

class Listing extends Model
{
    use HasFactory;

    protected $fillable = [
        'user_id', 'category_id', 'title', 'slug', 'description',
        'condition', 'price', 'negotiable', 'images', 'thumbnail',
        'tags', 'location', 'available_for_shipping', 'shipping_cost',
        'is_featured', 'is_active', 'status',
    ];

    protected $casts = [
        'price' => 'decimal:2',
        'shipping_cost' => 'decimal:2',
        'negotiable' => 'boolean',
        'is_featured' => 'boolean',
        'is_active' => 'boolean',
        'images' => 'array',
        'tags' => 'array',
    ];

    const STATUS_DRAFT = 'draft';
    const STATUS_PENDING_REVIEW = 'pending_review';
    const STATUS_ACTIVE = 'active';
    const STATUS_SOLD = 'sold';
    const STATUS_EXPIRED = 'expired';
    const STATUS_FLAGGED = 'flagged';
    const STATUS_REMOVED = 'removed';

    public function seller()
    {
        return $this->belongsTo(\App\Models\User::class, 'user_id');
    }

    public function category()
    {
        return $this->belongsTo(\App\Models\MarketplaceCategory::class, 'category_id');
    }

    public function orders()
    {
        return $this->hasMany(\App\Models\Marketplace\Order::class, 'listing_id');
    }

    public function reviews()
    {
        return $this->hasMany(\App\Models\Marketplace\Review::class);
    }

    public function favourites()
    {
        return $this->hasMany(\App\Models\Marketplace\Favourite::class, 'listing_id');
    }

    // Scopes
    public function scopeActive($query)
    {
        return $query->where('status', self::STATUS_ACTIVE);
    }

    public function scopeNotSold($query)
    {
        return $query->where('status', '!=', self::STATUS_SOLD);
    }

    public function scopeFeatured($query)
    {
        return $query->where('is_featured', true);
    }

    public function scopeCondition($query, string $condition)
    {
        return $query->where('condition', $condition);
    }

    public function scopeByCategory($query, int $categoryId)
    {
        return $query->where('category_id', $categoryId);
    }

    public function scopeBySeller($query, int $userId)
    {
        return $query->where('user_id', $userId);
    }

    public function scopeSearch($query, string $term)
    {
        return $query->where(function ($q) use ($term) {
            $q->where('title', 'like', "%{$term}%")
              ->orWhere('description', 'like', "%{$term}%")
              ->orWhereJsonContains('tags', $term);
        });
    }

    // Mutators
    protected static function booted(): void
    {
        static::creating(function (self $listing) {
            if (empty($listing->slug)) {
                $listing->slug = self::generateUniqueSlug($listing->title);
            }
        });
    }

    public static function generateUniqueSlug(string $title): string
    {
        $slug = Str::slug($title);
        $count = self::whereRaw("slug REGEXP '^{$slug}(-[0-9]+)?$'")->count();
        return $count > 0 ? "{$slug}-{$count}" : $slug;
    }

    public function getAverageRatingAttribute(): float
    {
        return $this->reviews()->avg('rating') ?? 0;
    }

    public function isFavouritedBy(\App\Models\User $user): bool
    {
        return $this->favourites()->where('user_id', $user->id)->exists();
    }
}
```

### 1.7 Transaction Workflow

```
1. Browse listings → Search/Filter → View detail
2. Add to Favourites (heart icon)
3. Click "Buy Now" → Select payment method (Wallet Escrow or External Card)
4. Payment processing:
   - Wallet Flow: Funds held in escrow via EscrowService
   - Card Flow: Redirected to Paystack/Kora/Stripe → Webhook funds escrow
5. Order status → pending → seller notified
6. Seller ships / arranges pickup → status = in_progress
7. Item delivered → status = delivered → buyer notified
8. Buyer confirms receipt:
   a. Confirm → Escrow released → Commission deducted → Net to seller wallet
   b. Raise Dispute → Escrow frozen → Dispute resolution process
9. Auto-release: If no buyer action within 7 days, escrow auto-releases to seller
```

## 3. Monetization Framework

### 3.1 Revenue Streams

| Stream | How It Works | Code Key | Config Key |
|--------|-------------|----------|------------|
| **Commission** | 5% deducted from seller payout on every completed order | `marketplace_commission_rate` | Default: 5% |
| **Featured Listing** | Seller pays to pin listing to top of search results for 7 days | `featured_listing_cost` | Default: ₦500 |
| **Promoted Listing** | Boosted visibility in search "Sponsored" section | `promoted_listing_cost` | Default: ₦200 |
| **Premium Seller Sub** | Monthly subscription: 2% commission (vs 5%), unlimited featured listings, analytics dashboard | `premium_seller_monthly` | Default: ₦2,000/mo |
| **Listing Boost** | One-time temporary ranking boost (24h) | `listing_boost_cost` | Default: ₦100 |
| **Urgent Badge** | "Urgent — Buy Now" flag on listing card | `urgent_listing_cost` | Default: ₦150 |
| **Advertising Banners** | Banner slots sold on marketplace homepage (CPM/flat) | Admin-managed | System settings |

### 3.2 Commission Calculation & Flow

```php
// app/Services/Marketplace/TransactionService.php (excerpt)
public function calculateCommission(float $orderTotal, bool $isPremiumSeller): array
{
    if ($isPremiumSeller) {
        $rate = (float) SystemSetting::get('marketplace_premium_commission_rate', 2);
    } else {
        $rate = (float) SystemSetting::get('marketplace_commission_rate', 5);
    }
    
    $commission = round($orderTotal * ($rate / 100), 2);
    $sellerPayout = round($orderTotal - $commission, 2);
    
    return [
        'total' => $orderTotal,
        'commission_rate' => $rate,
        'commission_amount' => $commission,
        'seller_payout' => $sellerPayout,
    ];
}
```

**Commission Transaction Records:**
- Type `marketplace_commission` → Platform revenue wallet
- Type `marketplace_payout` → Seller wallet (net of commission)
- All recorded in `financial_transactions` + `wallet_ledger`

### 3.3 Featured/Promoted Listing Logic

```php
// Add to ListingService
public function featureListing(Listing $listing, int $weeks = 1): bool
{
    $cost = (float) SystemSetting::get('featured_listing_cost', 500) * $weeks;
    
    $seller = $listing->seller;
    if (!$seller->wallet || !$seller->wallet->canWithdraw($cost)) {
        return false;
    }
    
    $seller->wallet->deductWithdrawable($cost, 'featured_listing');
    
    // Record financial transaction
    FinancialTransaction::create([
        'wallet_id' => $seller->wallet->id,
        'user_id' => $seller->id,
        'type' => 'featured_listing',
        'amount' => -$cost,
        'description' => "Featured listing: {$listing->title}",
        'reference' => 'FL-' . $listing->id,
        'status' => 'completed',
    ]);
    
    $listing->update([
        'is_featured' => true,
        'featured_until' => now()->addWeeks($weeks),
    ]);
    
    return true;
}
```

### 3.4 Premium Seller Subscription

```php
// app/Services/Marketplace/SubscriptionService.php (complete)
<?php

namespace App\Services\Marketplace;

use App\Models\Marketplace\Subscription;
use App\Models\User;
use App\Models\FinancialTransaction;
use App\Models\Wallet;
use Illuminate\Support\Facades\DB;

class SubscriptionService
{
    /**
     * Get premium subscription price.
     */
    public function getPrice(): float
    {
        return (float) SystemSetting::get('premium_seller_monthly', 2000);
    }

    /**
     * Subscribe a seller to premium tier.
     */
    public function subscribe(User $seller): array
    {
        DB::beginTransaction();
        try {
            $price = $this->getPrice();
            $wallet = $seller->wallet;

            if (!$wallet || !$wallet->canWithdraw($price)) {
                return ['success' => false, 'message' => 'Insufficient balance'];
            }

            // Create subscription record
            $subscription = Subscription::create([
                'user_id' => $seller->id,
                'plan' => 'premium',
                'amount_paid' => $price,
                'status' => 'active',
                'started_at' => now(),
                'expires_at' => now()->addMonth(),
            ]);

            // Deduct from wallet
            $wallet->deductWithdrawable($price, 'marketplace_subscription');
            $seller->update(['is_premium_seller' => true]);

            DB::commit();

            return [
                'success' => true,
                'message' => 'Subscribed to Premium Seller plan',
                'subscription' => $subscription,
                'expires_at' => $subscription->expires_at,
            ];
        } catch (\Exception $e) {
            DB::rollBack();
            return ['success' => false, 'message' => $e->getMessage()];
        }
    }

    /**
     * Cancel premium subscription.
     */
    public function cancel(User $seller): array
    {
        $activeSub = Subscription::where('user_id', $seller->id)
            ->where('status', 'active')
            ->first();

        if (!$activeSub) {
            return ['success' => false, 'message' => 'No active subscription found'];
        }

        $activeSub->update([
            'status' => 'cancelled',
            'cancelled_at' => now(),
            'cancellation_reason' => 'user_request',
        ]);

        $seller->update(['is_premium_seller' => false]);

        return ['success' => true, 'message' => 'Subscription cancelled'];
    }

    /**
     * Check if seller is premium.
     */
    public function isPremium(User $seller): bool
    {
        return Subscription::where('user_id', $seller->id)
            ->where('status', 'active')
            ->where('expires_at', '>', now())
            ->exists();
    }

    /**
     * Run monthly cron to renew/expire subscriptions.
     */
    public function processExpirations(): void
    {
        $expired = Subscription::where('status', 'active')
            ->where('expires_at', '<', now())
            ->get();

        foreach ($expired as $sub) {
            $sub->update(['status' => 'expired']);
            User::where('id', $sub->user_id)->update(['is_premium_seller' => false]);
        }
    }
}
```

```php
    const STATUS_PAID = 'paid';
    const STATUS_IN_PROGRESS = 'in_progress';
    const STATUS_DELIVERED = 'delivered';
    const STATUS_COMPLETED = 'completed';
    const STATUS_DISPUTED = 'disputed';
    const STATUS_CANCELLED = 'cancelled';
    const STATUS_REFUNDED = 'refunded';
    const STATUS_EXPIRED = 'expired';

    public function listing()

        return $this->belongsTo(Listing::class);
    }

    {
        return $this->belongsTo(\App\Models\User::class, 'buyer_id');
    }

    {
        return $this->belongsTo(\App\Models\User::class, 'seller_id');
    }

    public function escrow()
    {
        return $this->morphOne(\App\Models\EscrowTransaction::class, 'order');
    }

    public function disputes()
    {
        return $this->morphMany(\App\Models\Dispute::class, 'disputable');
    }

    // Scopes
    public function scopePending($query) { return $query->where('status', self::STATUS_PENDING); }
    public function scopePaid($query) { return $query->where('status', self::STATUS_PAID); }
    public function scopeInProgress($query) { return $query->where('status', self::STATUS_IN_PROGRESS); }
    public function scopeDelivered($query) { return $query->where('status', self::STATUS_DELIVERED); }
    public function scopeCompleted($query) { return $query->where('status', self::STATUS_COMPLETED); }
    public function scopeDisputed($query) { return $query->where('status', self::STATUS_DISPUTED); }
    public function scopeByBuyer($query, int $userId) { return $query->where('buyer_id', $userId); }
    public function scopeBySeller($query, int $userId) { return $query->where('seller_id', $userId); }

    {
        if ($this->status !== self::STATUS_PENDING) {
            return 'Order is not in pending state';
        }
```

```php
<?php

namespace App\Services\Marketplace;

use App\Models\Marketplace\Listing;
use App\Models\Marketplace\Favourite;
use App\Models\User;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;
use Illuminate\Database\Eloquent\Builder;

class ListingService
{
    /**
     * Create a new listing.
     */
    public function createListing(User $seller, array $data): Listing
    {
        return DB::transaction(function () use ($seller, $data) {
            $listing = Listing::create([
                'user_id' => $seller->id,
                'category_id' => $data['category_id'] ?? null,
                'title' => $data['title'],
                'description' => $data['description'],
                'price' => $data['price'],
                'condition' => $data['condition'] ?? 'good',
                'negotiable' => $data['negotiable'] ?? false,
                'images' => $data['images'] ?? [],
                'thumbnail' => $data['thumbnail'] ?? null,
                'tags' => $data['tags'] ?? [],
                'location' => $data['location'] ?? null,
                'available_for_shipping' => $data['available_for_shipping'] ?? false,
                'shipping_cost' => $data['shipping_cost'] ?? 0,
                'status' => $data['status'] ?? Listing::STATUS_DRAFT,
            ]);

            Log::info('Marketplace listing created', [
                'listing_id' => $listing->id,
                'seller_id' => $seller->id,
                'title' => $listing->title,
            ]);

            return $listing;
        });
    }

    /**
     * Update an existing listing.
     */
    public function updateListing(Listing $listing, array $data): bool
    {
        return DB::transaction(function () use ($listing, $data) {
            $updateData = [];
            $fillable = ['title', 'description', 'price', 'condition', 'negotiable',
                'thumbnail', 'tags', 'location', 'available_for_shipping',
                'shipping_cost', 'status'];

            foreach ($fillable as $field) {
                if (array_key_exists($field, $data)) {
                    $updateData[$field] = $data[$field];
                }
            }

            // Regenerate slug if title changed
            if (isset($data['title']) && $data['title'] !== $listing->title) {
                $updateData['slug'] = Listing::generateUniqueSlug($data['title']);
            }

            $listing->update($updateData);
            return true;
        });
    }

    /**
     * Publish/activate a listing.
     */
    public function publish(Listing $listing): bool
    {
        $listing->update([
            'status' => Listing::STATUS_ACTIVE,
            'is_active' => true,
        ]);
        return true;
    }

    /**
     * Unpublish/deactivate a listing.
     */
    public function unpublish(Listing $listing): bool
    {
        $listing->update(['status' => Listing::STATUS_DRAFT, 'is_active' => false]);
        return true;
    }

    /**
     * Mark listing as sold.
     */
    public function markAsSold(Listing $listing): bool
    {
        return DB::transaction(function () use ($listing) {
            $listing->update([
                'status' => Listing::STATUS_SOLD,
                'sold_at' => now(),
            ]);
            return true;
        });
    }

    /**
     * Toggle featured status.
     */
    public function toggleFeatured(Listing $listing): bool
    {
        $listing->update(['is_featured' => !$listing->is_featured]);
        return true;
    }

    /**
     * Search and filter listings.
     */
    public function search(array $filters): \Illuminate\Contracts\Pagination\LengthAwarePaginator
    {
        $query = Listing::query()
            ->where('status', Listing::STATUS_ACTIVE)
            ->where('is_active', true);

        // Search term
        if (!empty($filters['q'])) {
            $query->where(function (Builder $q) use ($filters) {
                $q->where('title', 'like', '%' . $filters['q'] . '%')
                  ->orWhere('description', 'like', '%' . $filters['q'] . '%')
                  ->orWhereJsonContains('tags', $filters['q']);
            });
        }

        // Category filter
        if (!empty($filters['category'])) {
            $query->where('category_id', $filters['category']);
        }

        // Condition filter
        if (!empty($filters['condition']) && is_array($filters['condition'])) {
            $query->whereIn('condition', $filters['condition']);
        }

        // Price range
        if (!empty($filters['min_price'])) {
            $query->where('price', '>=', $filters['min_price']);
        }
        if (!empty($filters['max_price'])) {
            $query->where('price', '<=', $filters['max_price']);
        }

        // University filter (same school)
        if (!empty($filters['university_id'])) {
            $query->where('user_id', function ($q) use ($filters) {
                $q->select('id')->from('users')
                  ->where('university_id', $filters['university_id']);
            });
        }

        // Shipping available
        if (!empty($filters['shipping'])) {
            $query->where('available_for_shipping', true);
        }

        // Sort
        $sort = $filters['sort'] ?? 'newest';
        switch ($sort) {
            case 'price_low':
                $query->orderBy('price', 'asc');
                break;
            case 'price_high':
                $query->orderBy('price', 'desc');
                break;
            case 'popular':
                $query->orderBy('favourites_count', 'desc');
                break;
            case 'featured':
                $query->orderBy('is_featured', 'desc')->orderBy('created_at', 'desc');
                break;
            default:
                $query->latest();
        }

        return $query->paginate($filters['per_page'] ?? 20);
    }
```

```php
            $totalBalance = $wallet->withdrawable_balance + $wallet->promo_credit_balance;
            if ($totalBalance < $total) {

            );

            if (!$escrowResult['success']) {
                throw new \Exception('Escrow hold failed: ' . $escrowResult['message']);

            // Update listing status
            $listing->update(['status' => Listing::STATUS_SOLD]);

            // Notify seller

                $seller,

                'cancelled_at' => now(),
```

```php
namespace App\Services\Marketplace;

use App\Models\FinancialTransaction;
use App\Models\WalletLedger;
use App\Models\Marketplace\Order;

class TransactionService
{
    /**
     * Calculate platform commission for an order.
     */
    public function calculateCommission(float $orderTotal, bool $isPremiumSeller = false): array
    {
        if ($isPremiumSeller) {
            $rate = (float) SystemSetting::get('marketplace_premium_commission_rate', 2);
        } else {
            $rate = (float) SystemSetting::get('marketplace_commission_rate', 5);
        }

        $commission = round($orderTotal * ($rate / 100), 2);
        $sellerPayout = round($orderTotal - $commission, 2);

        return [
            'total' => $orderTotal,
            'commission_rate' => $rate,
            'commission_amount' => $commission,
            'seller_payout' => $sellerPayout,
        ];
    }

    /**
     * Record commission transaction when escrow is released.
     */
    public function recordCommission(Order $order): void
    {
        $seller = $order->seller;
        $isPremium = $seller->subscriptions()
            ->where('status', 'active')
            ->where('expires_at', '>', now())
            ->exists();

        $calc = $this->calculateCommission($order->amount, $isPremium);

        // Record commission debit
        FinancialTransaction::create([
            'wallet_id' => $seller->wallet->id ?? null,
            'user_id' => $seller->id,
            'type' => 'marketplace_commission',
            'amount' => $calc['commission_amount'],
            'currency' => 'NGN',
            'status' => 'completed',
            'reference' => 'MKT-COMM-' . $order->id,
            'description' => "Marketplace commission for order #{$order->id}",
            'metadata' => [
                'order_id' => $order->id,
                'listing_id' => $order->listing_id,
                'commission_rate' => $calc['commission_rate'],
            ],
        ]);

        // Record seller payout
        FinancialTransaction::create([
            'wallet_id' => $seller->wallet->id ?? null,
            'user_id' => $seller->id,
            'type' => 'marketplace_payout',
            'amount' => $calc['seller_payout'],
            'currency' => 'NGN',
            'status' => 'completed',
            'reference' => 'MKT-PAYOUT-' . $order->id,
            'description' => "Marketplace seller payout for order #{$order->id}",
        ]);

        // Update seller rating
        $this->recalculateSellerRating($seller);
    }

    /**
     * Recalculate seller average rating.
     */
    public function recalculateSellerRating(User $seller): void
    {
        $avg = \App\Models\Marketplace\Review::where('reviewed_id', $seller->id)
            ->where('is_approved', true)
            ->avg('rating');

        $seller->update([
            'seller_rating' => round($avg ?? 0, 2),
            'seller_rating_count' => \App\Models\Marketplace\Review::where('reviewed_id', $seller->id)
                ->where('is_approved', true)
                ->count(),
        ]);
    }
```

```php
<?php

namespace App\Services\Marketplace;

use App\Models\User;
use App\Models\Wallet;
use App\Models\EscrowTransaction;
use App\Models\FinancialTransaction;
use App\Models\SystemSetting;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;
use Illuminate\Support\Str;

class MarketplaceService
{
    /**
     * Get commission rate setting.
     */
    public function getCommissionRate(): float
    {
        return (float) SystemSetting::get('marketplace_commission_rate', 5);
    }

    /**
     * Calculate escrow breakdown.
     */
    public function calculateEscrow(float $amount): array
    {
        $commissionRate = $this->getCommissionRate();
        $platformFee = round($amount * ($commissionRate / 100), 2);
        $escrowAmount = round($amount - $platformFee, 2);

        return [
            'amount' => $amount,
            'platform_fee' => $platformFee,
            'escrow_amount' => $escrowAmount,
            'commission_rate' => $commissionRate,
        ];
    }

    /**
     * Hold funds in escrow from buyer wallet.
     */
    public function holdInEscrow(User $buyer, User $seller, float $totalAmount, float $platformFee, $order, string $description): array
    {
        try {
            return DB::transaction(function () use ($buyer, $seller, $totalAmount, $platformFee, $order, $description) {
                $wallet = $buyer->wallet;

                if (!$wallet) {
                    return ['success' => false, 'message' => 'Wallet not found'];
                }

                $totalBalance = $wallet->withdrawable_balance + $wallet->promo_credit_balance;
                if ($totalBalance < $totalAmount) {
                    return ['success' => false, 'message' => 'Insufficient balance', 'available' => $totalBalance, 'required' => $totalAmount];
                }

                // Withdraw from buyer wallet (uses promo credit first, then withdrawable)
                if (!$wallet->addToEscrow($totalAmount)) {
                    return ['success' => false, 'message' => 'Failed to hold funds in escrow'];
                }

                $escrow = EscrowTransaction::create([
                    'transaction_no' => 'MKT-ESC-' . strtoupper(Str::random(10)),
                    'order_id' => $order->id,
                    'order_type' => get_class($order),
                    'payer_id' => $buyer->id,
                    'payee_id' => $seller->id,
                    'amount' => $totalAmount - $platformFee,
                    'platform_fee' => $platformFee,
                    'total_amount' => $totalAmount,
                    'status' => EscrowTransaction::STATUS_FUNDED,
                ]);

                Log::info('Marketplace escrow funded', [
                    'order_id' => $order->id,
                    'escrow_id' => $escrow->id,
                    'amount' => $totalAmount,
                    'buyer_id' => $buyer->id,
                ]);

                return ['success' => true, 'message' => 'Payment held in escrow', 'escrow_transaction' => $escrow];
            });
        } catch (\Exception $e) {
            Log::error('Marketplace escrow hold failed', [
                'buyer_id' => $buyer->id,
                'amount' => $totalAmount,
                'error' => $e->getMessage(),
            ]);
            return ['success' => false, 'message' => 'Failed to process payment: ' . $e->getMessage()];
        }
    }

    /**
     * Release escrow to seller on order completion.
     */
    public function releaseEscrow(EscrowTransaction $escrow, string $description): array
    {
        try {
            return DB::transaction(function () use ($escrow, $description) {
                $payerWallet = Wallet::firstOrCreate(
                    ['user_id' => $escrow->payer_id],
                    ['withdrawable_balance' => 0, 'promo_credit_balance' => 0,
                     'total_earned' => 0, 'total_spent' => 0, 'pending_balance' => 0, 'escrow_balance' => 0]
                );

                if ($payerWallet->escrow_balance < $escrow->total_amount) {
                    return ['success' => false, 'message' => 'Insufficient escrow balance'];
                }

                $payerWallet->escrow_balance = max(0, $payerWallet->escrow_balance - $escrow->total_amount);
                $payerWallet->save();

                $payeeWallet = Wallet::firstOrCreate(
                    ['user_id' => $escrow->payee_id],
                    ['withdrawable_balance' => 0, 'promo_credit_balance' => 0,
                     'total_earned' => 0, 'total_spent' => 0, 'pending_balance' => 0, 'escrow_balance' => 0]
                );

                // Net amount after deducting platform fee already recorded
                $payeeWallet->addWithdrawable($escrow->amount, 'marketplace_earning', $description);

                // Record earning in financial transactions
                FinancialTransaction::create([
                    'wallet_id' => $payeeWallet->id,
                    'user_id' => $escrow->payee_id,
                    'type' => 'marketplace_earning',
                    'amount' => $escrow->amount,
                    'status' => 'completed',
                    'reference' => 'MKT-EARN-' . $escrow->id,
                    'description' => "Earning from marketplace order — {$description}",
                ]);

                $escrow->status = EscrowTransaction::STATUS_RELEASED;
                $escrow->released_at = now();
                $escrow->save();

                Log::info('Marketplace escrow released', ['escrow_id' => $escrow->id]);

                return ['success' => true, 'message' => 'Payment released to seller', 'released_amount' => $escrow->amount];
            });
        } catch (\Exception $e) {
            Log::error('Marketplace escrow release failed', ['escrow_id' => $escrow->id, 'error' => $e->getMessage()]);
            return ['success' => false, 'message' => 'Failed to release payment: ' . $e->getMessage()];
        }
    }

    /**
     * Refund buyer from escrow.
     */
    public function refundFromEscrow(EscrowTransaction $escrow, string $description): array
    {
        try {
            return DB::transaction(function () use ($escrow, $description) {
                $payerWallet = Wallet::firstOrCreate(
                    ['user_id' => $escrow->payer_id],
                    ['withdrawable_balance' => 0, 'promo_credit_balance' => 0,
                     'total_earned' => 0, 'total_spent' => 0, 'pending_balance' => 0, 'escrow_balance' => 0]
                );

                if ($payerWallet->escrow_balance < $escrow->total_amount) {
                    return ['success' => false, 'message' => 'Insufficient escrow balance'];
                }

                $payerWallet->escrow_balance = max(0, $payerWallet->escrow_balance - $escrow->total_amount);
                $payerWallet->save();
                $payerWallet->addWithdrawable($escrow->total_amount, 'marketplace_refund', $description);

                FinancialTransaction::create([
                    'wallet_id' => $payerWallet->id,
                    'user_id' => $escrow->payer_id,
                    'type' => 'marketplace_refund',
                    'amount' => $escrow->total_amount,
                    'status' => 'completed',
                    'reference' => 'MKT-REFUND-' . $escrow->id,
                    'description' => "Refund: {$description}",
                ]);

                $escrow->status = EscrowTransaction::STATUS_CANCELLED;
                $escrow->save();

                return ['success' => true, 'message' => 'Refund processed', 'refunded_amount' => $escrow->total_amount];
            });
        } catch (\Exception $e) {
            Log::error('Marketplace escrow refund failed', ['escrow_id' => $escrow->id, 'error' => $e->getMessage()]);
            return ['success' => false, 'message' => 'Failed to process refund: ' . $e->getMessage()];
        }
    }

    /**
     * Get escrow transaction for an order.
     */
    public function getEscrowTransaction($order): ?EscrowTransaction
    {
        return EscrowTransaction::where('order_id', $order->id)
            ->where('order_type', get_class($order))
            ->first();
    }

    /**
     * Get platform fee for a specific order type.
     */
    public function getPlatformFee(float $amount, string $type = 'marketplace'): array
    {
        return $this->calculateEscrow($amount);
    }
}
```

```php
        $escrowAmount = round($amount - $platformFee, 2);

    /**
     * Hold funds in escrow from buyer wallet.
     */
    public function holdInEscrow(User $buyer, User $seller, float $totalAmount, float $platformFee, $order, string $description): array
    {
        try {
            return DB::transaction(function () use ($buyer, $seller, $totalAmount, $platformFee, $order, $description) {
                $wallet = $buyer->wallet;

                if (!$wallet) {
                    return ['success' => false, 'message' => 'Wallet not found'];
                }

                $totalBalance = $wallet->withdrawable_balance + $wallet->promo_credit_balance;
                if ($totalBalance < $totalAmount) {
                    return ['success' => false, 'message' => 'Insufficient balance', 'available' => $totalBalance, 'required' => $totalAmount];
                }

                // Withdraw from buyer wallet (uses promo credit first, then withdrawable)
                $remaining = $totalAmount;
                if ($wallet->promo_credit_balance > 0) {
                    $fromPromo = min($wallet->promo_credit_balance, $remaining);
                    $wallet->deductPromoCredit($fromPromo, 'marketplace_escrow', $description);
                    $remaining -= $fromPromo;
                }
                if ($remaining > 0) {
                    $wallet->deductWithdrawable($remaining, 'marketplace_escrow', $description);
                }

                // Increase escrow balance
                $wallet->escrow_balance = ($wallet->escrow_balance ?? 0) + $totalAmount;
                $wallet->save();

                $escrow = EscrowTransaction::create([
                    'transaction_no' => 'MKT-ESC-' . strtoupper(Str::random(10)),
                    'order_id' => $order->id,
                    'order_type' => get_class($order),
                    'payer_id' => $buyer->id,
                    'payee_id' => $seller->id,
                    'amount' => $totalAmount - $platformFee,
                    'platform_fee' => $platformFee,
                    'total_amount' => $totalAmount,
                    'status' => EscrowTransaction::STATUS_FUNDED,
                ]);

                Log::info('Marketplace escrow funded', [
                    'order_id' => $order->id,
                    'escrow_id' => $escrow->id,
                    'amount' => $totalAmount,
                    'buyer_id' => $buyer->id,
                ]);

                return ['success' => true, 'message' => 'Payment held in escrow', 'escrow_transaction' => $escrow];
            });
        } catch (\Exception $e) {
            Log::error('Marketplace escrow hold failed', [
                'buyer_id' => $buyer->id,
                'amount' => $totalAmount,
                'error' => $e->getMessage(),
            ]);
            return ['success' => false, 'message' => 'Failed to process payment: ' . $e->getMessage()];
        }
    }

    /**
     * Release escrow to seller on order completion.
     */
    public function releaseEscrow(EscrowTransaction $escrow, string $description): array
    {
        try {
            return DB::transaction(function () use ($escrow, $description) {
                $payerWallet = Wallet::firstOrCreate(
                    ['user_id' => $escrow->payer_id],
                    ['withdrawable_balance' => 0, 'promo_credit_balance' => 0,
                     'total_earned' => 0, 'total_spent' => 0, 'pending_balance' => 0, 'escrow_balance' => 0]
                );

                if ($payerWallet->escrow_balance < $escrow->total_amount) {
                    return ['success' => false, 'message' => 'Insufficient escrow balance'];
                }

                $payerWallet->escrow_balance -= $escrow->total_amount;
                $payerWallet->save();

                $payeeWallet = Wallet::firstOrCreate(
                    ['user_id' => $escrow->payee_id],
                    ['withdrawable_balance' => 0, 'promo_credit_balance' => 0,
                     'total_earned' => 0, 'total_spent' => 0, 'pending_balance' => 0, 'escrow_balance' => 0]
                );

                $payeeWallet->addWithdrawable($escrow->amount, 'marketplace_earning', $description);

                FinancialTransaction::create([
                    'wallet_id' => $payeeWallet->id,
                    'user_id' => $escrow->payee_id,
                    'type' => 'marketplace_earning',
                    'amount' => $escrow->amount,
                    'status' => 'completed',
                    'reference' => 'MKT-EARN-' . $escrow->id,
                    'description' => "Earning from marketplace order — {$description}",
                ]);

                $escrow->status = EscrowTransaction::STATUS_RELEASED;
                $escrow->released_at = now();
                $escrow->save();

                Log::info('Marketplace escrow released', ['escrow_id' => $escrow->id]);

                return ['success' => true, 'message' => 'Payment released to seller', 'released_amount' => $escrow->amount];
            });
        } catch (\Exception $e) {
            Log::error('Marketplace escrow release failed', ['escrow_id' => $escrow->id, 'error' => $e->getMessage()]);
            return ['success' => false, 'message' => 'Failed to release payment: ' . $e->getMessage()];
        }
    }

    /**
     * Refund buyer from escrow.
     */
    public function refundFromEscrow(EscrowTransaction $escrow, string $description): array
    {
        try {
            return DB::transaction(function () use ($escrow, $description) {
                $payerWallet = Wallet::firstOrCreate(
                    ['user_id' => $escrow->payer_id],
                    ['withdrawable_balance' => 0, 'promo_credit_balance' => 0,
                     'total_earned' => 0, 'total_spent' => 0, 'pending_balance' => 0, 'escrow_balance' => 0]
                );

                if ($payerWallet->escrow_balance < $escrow->total_amount) {
                    return ['success' => false, 'message' => 'Insufficient escrow balance'];
                }

                $payerWallet->escrow_balance -= $escrow->total_amount;
                $payerWallet->save();
                $payerWallet->addWithdrawable($escrow->total_amount, 'marketplace_refund', $description);

                FinancialTransaction::create([
                    'wallet_id' => $payerWallet->id,
                    'user_id' => $escrow->payer_id,
                    'type' => 'marketplace_refund',
                    'amount' => $escrow->total_amount,
                    'status' => 'completed',
                    'reference' => 'MKT-REFUND-' . $escrow->id,
                    'description' => "Refund: {$description}",
                ]);

                $escrow->status = EscrowTransaction::STATUS_CANCELLED;
                $escrow->save();

                return ['success' => true, 'message' => 'Refund processed', 'refunded_amount' => $escrow->total_amount];
            });
        } catch (\Exception $e) {
            Log::error('Marketplace escrow refund failed', ['escrow_id' => $escrow->id, 'error' => $e->getMessage()]);
            return ['success' => false, 'message' => 'Failed to process refund: ' . $e->getMessage()];
        }
    }

    /**
     * Get escrow transaction for an order.
     */
    public function getEscrowTransaction($order): ?EscrowTransaction
    {
        return EscrowTransaction::where('order_id', $order->id)
            ->where('order_type', get_class($order))
            ->first();
    }

    /**
     * Get platform fee for a specific amount.
     */
    public function getPlatformFee(float $amount): array
    {
        return $this->calculateEscrow($amount);
    }
}
```

### Phase 5: Payment Integration (Week 3)

**Flow A — Wallet Escrow (Primary):**
1. Buyer clicks "Buy Now" → `OrderController::createOrder()`
2. `OrderService::createOrder()` validates listing availability, calculates fee
3. `MarketplaceService::holdInEscrow()` withdraws from buyer wallet (promo + withdrawable)
4. Creates `EscrowTransaction` record
5. Order status → `pending`
6. Seller marks shipped → status → `in_progress`
7. Buyer marks delivered → status → `delivered`
8. Buyer confirms → `OrderService::confirmReceipt()` → `EscrowService::releaseEscrow()` + `TransactionService::recordCommission()`
9. Funds released to seller wallet minus commission

**Flow B — External Gateway (Paystack/Kora/Stripe):**
1. Buyer selects "Pay with Card" on checkout
2. `PaymentController::initialize()` redirects to gateway
3. Webhook callback (existing `/webhooks/paystack`, etc.) verifies and triggers order completion
4. On success: fund escrow via `MarketplaceService::holdInEscrow()` or direct payment marking

## 5. Safety and Trust Measures

### 5.1 Student Verification

| Method | Implementation |
|--------|---------------|
| **University email** | New `University` model has `domain` field. User submits email → system checks `@domain` against `universities.domain` → auto-verifies or sends confirmation code. New `UserVerification` type: `type = 'university'`. |
| **Government ID** | Existing `UserVerification` accepts `type = 'identity'` with `document_type`, `document_front`, `document_back`. Seller verification requires identity + address verification. |
| **Seller verification requirement** | Sellers must have at least one verified type (email or identity) before listings go live. Unverified sellers can create listings but status stays at `pending_review`. |

### 5.2 Escrow Payment Protection

- All marketplace payments go through **escrow** (existing `EscrowTransaction` model, new `order_type = marketplace`)
- Funds released only after buyer confirms receipt or auto-release timer (7 days) expires
- Dispute freezes escrow until resolution
- Refunds return to buyer's withdrawable balance
- Escrow balance tracked separately in `wallets.escrow_balance`

### 5.3 Dispute Resolution

Reuses existing `Dispute` model with `disputable` polymorphic morph:

- Buyer or seller opens dispute → order status = `disputed`
- 48-hour response window for respondent
- Admin resolution options: `buyer_wins` (refund from escrow to buyer), `seller_wins` (release escrow to seller), `split` (partial refund + partial release)
- `MarketplaceService::resolveDispute()` handles fund movement on resolution
- All dispute evidence stored as JSON

### 5.4 Content Moderation

- Listings start as `draft` → `pending_review` → `active`
- Admin review queue for flagged/inappropriate content
- Automated keyword filter for prohibited items (add to `SystemSetting` as blocked keywords list)
- User reporting mechanism: flag listing button → creates `FlaggedListing` record (can be added as future enhancement)
- Three strikes policy: seller with 3 rejected listings gets account warning

### 5.5 Fraud Prevention

- **Rate limiting:** Max 5 listings/hour for new users (< 30 days), 20/hour for established users. Use Laravel's `throttle` middleware.
- **Duplicate detection:** Check for listings with >90% title similarity + same price within 24 hours
- **Suspicious pattern alerts:** Flag users creating >10 orders/day (potential fake transactions) via `FraudLog` model
- **Transaction idempotency:** Use existing `IdempotencyKey` model; all state-changing operations require idempotency key
- **Admin override:** Admin can suspend sellers, remove listings, force-refund orders

```php
    public function index(Request $request)

        return back()->with('success', 'Listing approved');
    }

    public function reject(Listing $listing, Request $request)
    {
        $request->validate(['reason' => 'required|string|min:10']);

            'rejection_reason' => $request->reason,
        ]);
        notifyUser($listing->seller, 'Listing Rejected',
            "Your listing \"{$listing->title}\" was rejected: {$request->reason}");
        return back()->with('success', 'Listing rejected');
    }

    public function flag(Listing $listing)
    {
        $listing->update(['status' => Listing::STATUS_FLAGGED]);
        return back()->with('success', 'Listing flagged for review');
    }

    public function feature(Listing $listing)
    {
        $listing->update(['is_featured' => true, 'featured_until' => now()->addDays(7)]);
        return back()->with('success', 'Listing featured for 7 days');
    }

    public function unfeature(Listing $listing)
    {
        $listing->update(['is_featured' => false, 'featured_until' => null]);
        return back()->with('success', 'Listing unfeatured');
    }

    // ── Dispute Resolution ────────────────────────────────

    public function disputes()
    {
        $disputes = Dispute::with('complainant', 'respondent', 'disputable')
            ->latest()
            ->paginate(20);
        return view('admin.marketplace.disputes', compact('disputes'));
    }

    public function resolveDispute(Dispute $dispute, Request $request)
    {
        $request->validate([
            'resolution' => 'required|in:buyer_wins,seller_wins,split',
            'notes' => 'required|string|min:20',
        ]);

        // Delegates to MarketplaceService::resolveDispute()
        $result = app(\App\Services\Marketplace\MarketplaceService::class)
            ->resolveDispute($dispute, $request->resolution, $request->notes);

        if ($result['success']) {
            return back()->with('success', 'Dispute resolved');
        }
        return back()->with('error', $result['message']);
    }

    // ── Revenue Dashboard ─────────────────────────────────

    public function revenue()
    {
        $totalRevenue = FinancialTransaction::where('type', 'marketplace_commission')
            ->sum('amount');
        $totalPayouts = FinancialTransaction::where('type', 'marketplace_payout')
            ->sum('amount');
        $totalRefunds = FinancialTransaction::where('type', 'marketplace_refund')
            ->sum('amount');
        $monthlyRevenue = FinancialTransaction::where('type', 'marketplace_commission')
            ->whereMonth('created_at', now()->month)
            ->sum('amount');

        $topSellers = User::withSum('marketplaceOrders as revenue', function ($q) {
            $q->completed();
        })->orderByDesc('revenue')->limit(10)->get();

        return view('admin.marketplace.revenue', compact(
            'totalRevenue', 'totalPayouts', 'totalRefunds',
            'monthlyRevenue', 'topSellers'
        ));
    }

    // ── Seller Management ──────────────────────────────────

    public function sellers()
    {
        $sellers = User::whereHas('marketplaceListings')
            ->withCount('marketplaceListings')
            ->withCount('marketplaceReviews')
            ->latest()
            ->paginate(30);
        return view('admin.marketplace.sellers', compact('sellers'));
    }

    public function verifySeller(User $user)
    {
        $user->update(['marketplace_seller_verified' => true]);
        notifyUser($user, 'Seller Verified',
            'Your seller account has been verified!');
        return back()->with('success', 'Seller verified');
    }

    public function suspendSeller(User $user)
    {
        $user->marketplaceListings()->update(['status' => 'removed']);
        $user->update(['marketplace_seller_verified' => false]);
        return back()->with('success', 'Seller suspended');
    }
}
```

## 7. Safety and Trust Measures

### 7.1 Student Verification

| Method | Implementation |
|--------|---------------|
| **University email** | `University` model stores known education domains (`@unilorin.edu.ng`, etc.). `App\Models\University::findByEmailDomain($email)` checks domain. `UserVerification::create(['type' => 'university', 'status' => 'verified'])` marks verified. Only verified-university users appear in "Same School" filters. |
| **Government ID** | Existing `UserVerification` with `type = 'identity'` handles national ID, driver's license, or student card uploads. Required for premium seller status. |
| **Seller requirements** | Seller verification = email verified **OR** identity verified. Unverified sellers get `draft`-only listings (buyers can't see them). |

### 7.2 Escrow Payment Protection

- 100% of marketplace payments route through **escrow** — buyer wallet → escrow balance → release on confirmation
- `EscrowTransaction` records every state transition with timestamps
- Auto-release after configurable days (default: 7) prevents indefinite fund holds
- Dispute freezes escrow; resolution determines fund direction
- Refunds go back to buyer's withdrawable balance

### 7.3 Dispute Resolution Protocol

1. **Opening:** Buyer or seller submits dispute with title, description, evidence (photos, messages)
2. **Review:** Admin reviews within 48 hours; order and escrow frozen
3. **Resolution options:**
   - `buyer_wins` — full refund from escrow to buyer
   - `seller_wins` — full release from escrow to seller
   - `split` — partial refund + partial release (manual admin adjustment)
4. **Appeal:** Second-level review by senior moderator
5. **All outcomes:** Recorded in `financial_transactions` with full audit trail

### 7.4 Content Moderation

- **Pre-publication review:** Listings require admin approval before going live
- **Automated checks:** Title/description filtered against blocked keywords list (`SystemSetting` key: `marketplace_blocked_keywords`)
- **User reporting:** Report button on listings; triggers review flag
- **Three-strike policy:** 3 rejected listings = 24-hour listing suspension
- **Image moderation:** Check image hashes against known violations database

### 7.5 Fraud Prevention

- **Rate limiting:** Laravel throttle — new users: 5 listings/hour, verified: 20/hour
- **Duplicate detection:** Cron job checks listings with >90% Jaccard similarity on title+price within 24h
- **Fake order detection:** Flag accounts with >10 orders to same user in 24h → auto-hold pending review
- **Idempotency:** `IdempotencyKey` used on all checkout/order operations
- **Earnings pattern analysis:** `FraudLog` integration for suspicious earning spikes

```nginx
server {

        fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
        fastcgi_intercept_errors off;
        fastcgi_buffer_size 16k;
        fastcgi_buffers 4 16k;
    }

    location ~ /\.(?!well-known).* {
        deny all;
    }
}
```
